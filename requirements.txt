# Runtime dependencies.
#
# `openai==0.28.1` used to head this list and has been removed: nothing in the
# project imports it. The LLM client here is Groq (`llm_file_assistant.py:4`).
# A dependency nobody imports is not free — it gets installed, it gets scanned by
# every vulnerability tool, and it misleads the next reader about which provider
# this project actually talks to. That particular pin was also pre-1.0 `openai`,
# whose API was reshaped entirely in 1.0, so anyone taking the hint would have
# written code against an interface this project never used.

# ── LLM client and configuration ─────────────────────────────────────────────
groq==1.7.0
python-dotenv==1.2.4

# ── Document parsing (imported lazily inside fs_tools.read_file) ─────────────
# Imported at call time rather than module level so a missing parser degrades to
# "PDF support unavailable" for that one file instead of stopping the whole app
# from starting. The tests therefore need neither of these installed.
PyPDF2==3.0.1
python-docx==0.8.11

# ── Retrieval / RAG ──────────────────────────────────────────────────────────
# Pinned to versions that install on Python 3.12. The previous pins (numpy
# 1.21.6, pandas 1.3.5, scikit-learn 1.0.2) predate 3.12 and do not install on it.
#
# chromadb is the baseline store, and its bundled all-MiniLM-L6-v2 (ONNX) is
# also the embedding model for the Postgres store, so the two are compared on
# identical vectors. sentence-transformers was listed but never imported.
chromadb==1.5.9
numpy==2.5.5
scikit-learn==1.9.1          # TF-IDF fallback in resume_rag.py

# Postgres + pgvector store (pg_store.py)
psycopg[binary]==3.3.9
pgvector==0.5.1              # numpy <-> vector adapter for psycopg
# Embedded Postgres 16 + pgvector 0.6.2, used only when DATABASE_URL is unset:
# lets the store run on a machine without Docker or admin rights.
# No wheel for ARM Linux: set DATABASE_URL there (e.g. docker compose up -d).
pgserver==0.1.4; sys_platform != "linux" or platform_machine == "x86_64"

# ── Agent graph (graph_agent.py) ─────────────────────────────────────────────
# LangGraph for two things only: interrupt (approval before write_file) and the
# Postgres checkpointer (memory across restarts). langchain-core comes with it
# for message types; langchain-groq is the chat-model wrapper LangGraph calls.
langgraph==1.2.24
langgraph-checkpoint-postgres==3.2.1
langchain-groq==1.2.1

# ── Analysis notebook (rag_analysis.ipynb) ───────────────────────────────────
pandas==3.1.1
matplotlib==3.11.2
jupyter
